cmake_minimum_required(VERSION 3.16)
project(ros_leg_tracking VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_ROS "Build ROS Noetic adapter" OFF)
option(ENABLE_SANITIZERS "Address and undefined behavior sanitizers" OFF)
include(FetchContent)
find_package(Eigen3 QUIET)
if(NOT Eigen3_FOUND)
  FetchContent_Declare(eigen URL https://github.com/eigenteam/eigen-git-mirror/archive/refs/tags/3.3.7.tar.gz)
  FetchContent_GetProperties(eigen)
  if(NOT eigen_POPULATED)
    FetchContent_Populate(eigen)
  endif()
  add_library(Eigen3::Eigen INTERFACE IMPORTED)
  set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${eigen_SOURCE_DIR}")
endif()
add_library(leg_tracking src/perception.cpp src/tracking.cpp src/control.cpp)
target_include_directories(leg_tracking PUBLIC include)
target_link_libraries(leg_tracking PUBLIC Eigen3::Eigen)
if(MSVC)
  target_compile_options(leg_tracking PRIVATE /W4)
else()
  target_compile_options(leg_tracking PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ENABLE_SANITIZERS)
  target_compile_options(leg_tracking PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(leg_tracking PUBLIC -fsanitize=address,undefined)
endif()
add_executable(leg_cli src/cli.cpp)
target_link_libraries(leg_cli PRIVATE leg_tracking)
add_executable(feature_benchmark src/benchmark.cpp)
target_link_libraries(feature_benchmark PRIVATE leg_tracking)
include(CTest)
if(BUILD_TESTING)
  if(MINGW)
    add_compile_definitions(GTEST_HAS_PTHREAD=0 GTEST_IS_THREADSAFE=0)
  endif()
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.tar.gz)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
  if(TARGET GTest::Main AND NOT TARGET GTest::gtest_main)
    add_library(GTest::gtest_main ALIAS GTest::Main)
  endif()
  add_executable(unit_tests tests/unit_tests.cpp)
  target_link_libraries(unit_tests PRIVATE leg_tracking GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(unit_tests)
endif()
if(BUILD_ROS)
  find_package(catkin REQUIRED COMPONENTS roscpp rospy sensor_msgs geometry_msgs visualization_msgs tf2_ros tf2_geometry_msgs)
  catkin_package(INCLUDE_DIRS include LIBRARIES leg_tracking CATKIN_DEPENDS roscpp sensor_msgs geometry_msgs visualization_msgs tf2_ros tf2_geometry_msgs)
  add_executable(leg_tracking_node ros/node.cpp)
  target_include_directories(leg_tracking_node PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(leg_tracking_node PRIVATE leg_tracking ${catkin_LIBRARIES})
  catkin_install_python(PROGRAMS tests/ros_integration.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  if(CATKIN_ENABLE_TESTING)
    find_package(rostest REQUIRED)
    add_rostest(tests/integration.test)
  endif()
  install(TARGETS leg_tracking leg_tracking_node leg_cli ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
  install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
endif()
